  本项目构建一个视频评论智能分析平台，支持B站/抖音评论自动采集、清洗、
  DeepSeek大模型话题分类与情感分析、Qwen3-Embedding向量聚类分析、
  Chart.js数据可视化、SSE实时进度推送、实时追踪与邮件报告等核心功能。

  技术栈：
    后端：Python 3.10 + FastAPI + SQLAlchemy 2.0 (异步) + MySQL 8.0
    前端：Vue.js 3 (Composition API) + Chart.js + 原生 JS (SPA)
    AI：  DeepSeek (大模型分析) + Qwen3-Embedding-0.6B (向量聚类)
    部署：Uvicorn ASGI + 阿里云 DirectMail

    backend/              后端代码

    level=logging.DEBUG,
    format="[%(asctime)s] %(levelname)-7s %(name)s | %(message)s",
    datefmt="%H:%M:%S",
)
logger = logging.getLogger("main")


@asynccontextmanager
async def lifespan(app: FastAPI):
    """
    FastAPI 生命周期管理（替代已废弃的 on_event）
    启动时: 初始化数据库引擎 → 恢复活跃的追踪任务
    关闭时: 停止所有追踪轮询 → 释放数据库连接池
    """
    logger.info("正在启动服务...")
    await init_db()  # 创建数据库引擎 + 自动建表
    logger.info("数据库初始化完成")

    # 恢复活跃的追踪任务（服务重启后自动恢复）
    try:
        from backend.service.tracking_service import resume_all_active_trackings
        resumed = await resume_all_active_trackings()
        if resumed:
            logger.info("已恢复 %d 个活跃追踪任务", resumed)
    except Exception as e:
        logger.error("恢复追踪任务失败: %s", e)

    yield  # ← 此处挂起，等待应用关闭

    # 关闭时优雅停止所有后台追踪轮询
    try:
        from backend.service.tracking_service import shutdown_all_trackings
        await shutdown_all_trackings()
    except Exception as e:
        logger.error("关闭追踪任务失败: %s", e)

    await close_db()  # 释放数据库连接池
    logger.info("数据库连接池已释放")

    lifespan=lifespan,  # 使用新的 lifespan 方式管理生命周期
)

# ---------- 请求日志中间件 ----------
@app.middleware("http")
async def log_requests(request: Request, call_next):
    """
    记录每个 HTTP 请求的方法、路径、响应状态码和耗时（毫秒）
    用于调试和性能监控
    """
    start = time.time()
    response = await call_next(request)
    elapsed = (time.time() - start) * 1000
    logger.debug("%s %s → %d (%.1fms)",
                 request.method, request.url.path,
                 response.status_code, elapsed)
    return response


# ---------- 统一异常处理（三层兜底）----------
# 所有异常统一返回 {"code": xxx, "message": "...", "data": null}

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """Pydantic 请求参数校验失败 → 422"""
    errors = exc.errors()
    first_msg = errors[0]["msg"] if errors else "请求参数校验失败"
    return JSONResponse(
        status_code=422,
        content={"code": 422, "message": first_msg, "data": None},
    )


@app.exception_handler(HTTPException)
async def http_exception_handler(request: Request, exc: HTTPException):
    """业务逻辑主动抛出的 HTTP 异常（如 401/403/409）"""
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": exc.status_code, "message": exc.detail, "data": None},
    )


@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """兜底异常处理 → 500 服务器内部错误"""
    logger.error("未处理异常 | %s %s — %s",
                 request.method, request.url.path, str(exc), exc_info=True)
    return JSONResponse(
        status_code=500,
        content={"code": 500, "message": "服务器内部错误", "data": None},
    )

@app.get("/")
async def root():
    """根路径 → 重定向到登录页"""
    return RedirectResponse(url="/login.html")


@app.get("/{filename:path}")
async def serve_frontend(filename: str):
    """
    托管前端静态资源（HTML、图片等）
    重要: API 路由注册在前，优先匹配，静态资源作为兜底
    包含路径遍历攻击防护（防止 ../../etc/passwd 类型攻击）
    """
    safe_path = os.path.normpath(os.path.join(frontend_dir, filename))
    if not safe_path.startswith(frontend_dir):
        raise HTTPException(status_code=404, detail="页面不存在")

    if os.path.isfile(safe_path):
        resp = FileResponse(safe_path)
        # 开发阶段禁用 JS/CSS/HTML 缓存，方便修改即时生效
        if safe_path.endswith(('.js', '.css', '.html')):
            resp.headers['Cache-Control'] = 'no-cache, no-store, must-revalidate'
        return resp

    if os.path.isdir(safe_path):
        index_path = os.path.join(safe_path, "index.html")
        if os.path.isfile(index_path):
            return FileResponse(index_path)

    raise HTTPException(status_code=404, detail="页面不存在")

三、数据库引擎 — 异步懒加载单例 (backend/database.py)
================================================================================

"""SQLAlchemy 2.0 异步引擎 + Session 工厂 + ORM Base 类

设计要点：
  1. 懒加载单例模式：引擎不在模块导入时创建，而是在 FastAPI lifespan 中
     调用 init_db() 时首次创建。这解决了 uvicorn reload 子进程环境下
     conda 环境找不到 asyncmy 驱动的问题。
  2. 连接池配置：pool_size=10, max_overflow=20，适合中等并发场景。
  3. get_db() 作为 FastAPI 依赖注入，自动管理 session 生命周期和事务。
"""

from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy.orm import DeclarativeBase
from backend.config import DATABASE_URL

# 懒加载 — 只在 lifespan 中 init_db() 后才可用
_engine = None
_AsyncSessionLocal = None


def get_engine():
    """获取 SQLAlchemy 异步引擎（懒加载单例）"""
    global _engine
    if _engine is None:
        _engine = create_async_engine(
            DATABASE_URL,
            pool_size=10,        # 连接池核心连接数
            max_overflow=20,     # 最大溢出连接数
            echo=False           # 不打印 SQL（生产环境）
        )
    return _engine


def get_sessionmaker():
    """获取异步 Session 工厂（懒加载单例）"""
    global _AsyncSessionLocal
    if _AsyncSessionLocal is None:
        _AsyncSessionLocal = async_sessionmaker(
            get_engine(),
            class_=AsyncSession,
            expire_on_commit=False  # commit 后不使对象过期，方便后续访问属性
        )
    return _AsyncSessionLocal


async def init_db():
    """在 lifespan startup 中调用，确保引擎就绪 + 自动建表"""
    engine = get_engine()
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)  # 自动创建所有表


async def close_db():
    """在 lifespan shutdown 中调用，释放连接池资源"""
    global _engine
    if _engine is not None:
        await _engine.dispose()
        _engine = None


class Base(DeclarativeBase):
    """所有 ORM 模型的基类"""
    pass


async def get_db():
    """
    每个 HTTP 请求注入一个数据库 session
    自动管理事务: 正常 → commit，异常 → rollback，最后 → close

    使用方式（在路由函数中）：
      @router.get("/xxx")
      async def xxx(db: AsyncSession = Depends(get_db)):
          ...
    """
    async with get_sessionmaker()() as session:
        try:
            yield session
            await session.commit()     # 请求正常完成 → 提交事务
        except Exception:
            await session.rollback()   # 异常 → 回滚
            raise
        finally:
            await session.close()      # 无论如何 → 关闭 session

四、JWT 鉴权依赖注入 (backend/dependencies.py)
================================================================================

"""FastAPI 依赖注入 — JWT 鉴权中间件

所有需要登录的接口只需在参数中声明:
    current_user: User = Depends(get_current_user)
即可自动完成 JWT 解析、用户查询、权限校验。
"""

from fastapi import Depends, HTTPException, Header
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from jose import jwt, JWTError
from backend.config import JWT_SECRET, JWT_ALGORITHM
from backend.database import get_db


async def get_current_user(
    authorization: str = Header(default=""),
    db: AsyncSession = Depends(get_db),
):
    """
    从 HTTP 请求头 Authorization: Bearer <token> 中解析 JWT

    校验流程:
      1. 提取 Bearer 后的 token 字符串
      2. 用 JWT_SECRET 解密验证签名 + 过期时间
      3. 从 payload.sub 获取 user_id
      4. 查询数据库确认用户存在
      5. 返回完整 User ORM 对象供下游使用

    任何一步失败 → 抛出 401 Unauthorized
    前端 api.js 统一拦截 401 响应，清除本地 token 并重定向到登录页
    """
    if not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="未登录")

    token = authorization[7:]  # 截掉 "Bearer " 前缀

    try:
        payload = jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        user_id: str = payload.get("sub")
        if not user_id:
            raise HTTPException(status_code=401, detail="无效的 Token")
    except JWTError:
        raise HTTPException(status_code=401, detail="Token 已过期或无效")

    from backend.models.user import User
    result = await db.execute(select(User).where(User.id == user_id))
    user = result.scalar_one_or_none()
    if not user:
        raise HTTPException(status_code=401, detail="用户不存在")

    return user

--- 用户表 (backend/models/user.py) ---

"""用户表 — 对应 MySQL users 表

字段说明:
  - id:           UUID 主键
  - username:     用户名（唯一，3-64 位）
  - password_hash: bcrypt 加密后的密码哈希（不可逆）
  - email:        邮箱（唯一，用于找回密码 + 接收分析报告）
  - role:         角色枚举 (admin=管理员 / analyst=分析师 / viewer=观察者)
  - credits:      积分余额（用于实时追踪消耗）
  - api_key:      用户 API Key（预留功能）
  - last_login_at: 最后登录时间
"""

import uuid
import enum
from datetime import datetime
from sqlalchemy import String, DateTime, Integer, Enum as SAEnum
from sqlalchemy.orm import Mapped, mapped_column
from backend.database import Base


class UserRole(str, enum.Enum):
    """三级角色权限"""
    admin = "admin"      # 管理员
    analyst = "analyst"  # 分析师
    viewer = "viewer"    # 观察者


class User(Base):
    __tablename__ = "users"

    id: Mapped[str] = mapped_column(
        String(36), primary_key=True, default=lambda: str(uuid.uuid4())
    )
    username: Mapped[str] = mapped_column(
        String(64), unique=True, nullable=False
    )
    password_hash: Mapped[str] = mapped_column(
        String(256), nullable=False
    )
    email: Mapped[str | None] = mapped_column(
        String(128), unique=True, nullable=True
    )
    avatar_url: Mapped[str | None] = mapped_column(
        String(512), nullable=True
    )
    role: Mapped[UserRole] = mapped_column(
        SAEnum(UserRole), default=UserRole.analyst, nullable=False
    )
    api_key: Mapped[str | None] = mapped_column(String(256), nullable=True)
    credits: Mapped[int] = mapped_column(
        Integer, default=0, nullable=False
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime, default=datetime.utcnow, nullable=False
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime, default=datetime.utcnow, onupdate=datetime.utcnow
    )
    last_login_at: Mapped[datetime | None] = mapped_column(
        DateTime, nullable=True
    )


--- 分析任务表 (backend/models/analysis_task.py) ---

"""分析任务表 — 对应 MySQL analysis_tasks 表

每条记录记录一次视频评论分析任务的全部配置和运行状态。

核心字段:
  - status: 九阶段状态流转:
      queued → collecting → cleaning → embedding → clustering
      → topic_gen → summarizing → completed / failed
  - mode: normal（普通分析）或 tracking（实时追踪）
  - analysis_method: llm（DeepSeek 大模型）或 cluster（向量聚类）
  - progress_pct: 前端轮询展示的进度百分比 (0-100)
  - error_message: JSON 格式，存储清洗统计 + AI 综述 + 属性情感数据
  - total_comments_processed: 已处理评论总数（追踪模式下会持续增长）
"""


class AnalysisMode(str, enum.Enum):
    normal = "normal"      # 一次性分析
    tracking = "tracking"  # 持续追踪


class AnalysisStatus(str, enum.Enum):
    """Pipeline 九阶段状态机"""
    queued = "queued"          # 排队中
    collecting = "collecting"  # 正在抓取评论
    cleaning = "cleaning"      # 正在清洗数据
    embedding = "embedding"    # 正在向量化
    clustering = "clustering"  # 正在聚类
    topic_gen = "topic_gen"    # 正在生成话题
    summarizing = "summarizing" # 正在 AI 分析/综述
    completed = "completed"    # 已完成
    failed = "failed"          # 失败


class AnalysisTask(Base):
    __tablename__ = "analysis_tasks"

    id: Mapped[str] = mapped_column(
        String(36), primary_key=True, default=lambda: str(uuid.uuid4())
    )
    user_id: Mapped[str] = mapped_column(
        String(36), nullable=False, comment="创建者 ID"
    )
    video_id: Mapped[str] = mapped_column(
        String(36), nullable=False, comment="目标视频 ID"
    )
    platform: Mapped[str] = mapped_column(
        String(16), nullable=False, comment="平台: bilibili / douyin"
    )
    mode: Mapped[AnalysisMode] = mapped_column(
        SAEnum(AnalysisMode), nullable=False
    )
    comment_limit: Mapped[int] = mapped_column(
        Integer, nullable=False, default=500, comment="评论抓取上限 (0=全部)"
    )
    status: Mapped[AnalysisStatus] = mapped_column(
        SAEnum(AnalysisStatus), nullable=False, default=AnalysisStatus.queued
    )
    progress_pct: Mapped[int] = mapped_column(
        Integer, nullable=False, default=0, comment="完成百分比 0-100"
    )
    total_comments_processed: Mapped[int] = mapped_column(
        Integer, nullable=False, default=0
    )
    topic_count: Mapped[int] = mapped_column(
        Integer, nullable=False, default=0, comment="发现的 Topic 数量"
    )
    error_message: Mapped[str | None] = mapped_column(
        Text, nullable=True, comment="JSON: 清洗统计+AI综述+属性情感"
    )
    analysis_method: Mapped[str] = mapped_column(
        String(8), nullable=False, default="llm", comment="分析方法: llm | cluster"
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime, nullable=False, default=datetime.utcnow
    )
    completed_at: Mapped[datetime | None] = mapped_column(
        DateTime, nullable=True
    )

六、认证服务 — 密码哈希 + JWT 签发 (backend/service/auth_service.py)

"""认证服务 — bcrypt 密码哈希 + JWT 令牌生成

安全设计:
  - 密码使用 bcrypt 加盐哈希存储（不可逆，防彩虹表）
  - JWT 载荷包含 user_id (sub) + 签发时间 (iat) + 过期时间 (exp)
  - "记住我"功能: expire_seconds=30天，否则默认 24 小时
"""

from datetime import datetime, timedelta, timezone
import bcrypt
from jose import jwt
from backend.config import JWT_SECRET, JWT_ALGORITHM, JWT_EXPIRE_SECONDS


def hash_password(password: str) -> str:
    """
    对明文密码进行 bcrypt 哈希
    bcrypt 自动生成随机盐值并内嵌到哈希结果中，每次哈希结果不同
    """
    return bcrypt.hashpw(
        password.encode("utf-8"), bcrypt.gensalt()
    ).decode("utf-8")


def verify_password(password: str, password_hash: str) -> bool:
    """
    验证明文密码与 bcrypt 哈希是否匹配
    bcrypt.checkpw 自动从哈希中提取盐值进行比对
    """
    return bcrypt.checkpw(
        password.encode("utf-8"), password_hash.encode("utf-8")
    )


def create_access_token(user_id: str, expire_seconds: int | None = None) -> str:
    """
    签发 JWT 访问令牌

    Args:
        user_id: 用户 ID（存入 sub 字段）
        expire_seconds: 自定义过期秒数。
                        None → 使用全局默认 24 小时
                        30*24*3600 → "记住我" 30 天
    """
    seconds = expire_seconds if expire_seconds is not None else JWT_EXPIRE_SECONDS
    expire = datetime.now(timezone.utc) + timedelta(seconds=seconds)
    payload = {
        "sub": user_id,                     # 主题 = 用户 ID
        "exp": expire,                      # 过期时间
        "iat": datetime.now(timezone.utc),  # 签发时间
    }
    return jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM)

"""认证接口 — /api/auth/*

# DB 角色枚举 → 前端中文显示名
_ROLE_LABELS = {"admin": "管理员", "analyst": "分析师", "viewer": "观察者"}


def _user_to_frontend(user: User) -> dict:
    """将 User ORM 对象转为前端需要的格式（密码不出现在响应中）"""
    return {
        "user_id": user.id,
        "username": user.username,
        "role": _ROLE_LABELS.get(user.role.value, user.role),
        "avatar_initial": user.username[0].upper() if user.username else "U",
        "email": user.email,
    }


@router.post("/login")
async def login(body: LoginRequest, db: AsyncSession = Depends(get_db)):
    """
    用户登录

    安全设计:
      - 账号不存在和密码错误返回相同的错误提示，防止用户名枚举攻击
      - 登录失败记录 warning 日志，便于安全审计
      - 登录成功更新 last_login_at 字段
    """
    result = await db.execute(
        select(User).where(User.username == body.username)
    )
    user = result.scalar_one_or_none()

    if not user or not verify_password(body.password, user.password_hash):
        # 统一错误提示，不区分"用户名不存在"和"密码错误"
        raise HTTPException(status_code=401, detail="用户名或密码错误")

    user.last_login_at = datetime.now(timezone.utc)
    await db.flush()

    # 签发 token: "记住我" → 30 天; 否则 → 默认 24 小时
    token = create_access_token(
        user.id,
        30 * 24 * 3600 if body.remember else None
    )
    return ok({"token": token, "user": _user_to_frontend(user)})


@router.post("/register")
async def register(body: RegisterRequest, db: AsyncSession = Depends(get_db)):
    """
    用户注册

    校验规则:
      - 用户名 ≥ 3 位
      - 密码 ≥ 6 位
      - 两次密码必须一致
      - 用户名和邮箱不可重复（409 Conflict）
    """
    if body.password != body.confirm_password:
        raise HTTPException(status_code=422, detail="两次密码不一致")
    if len(body.username) < 3:
        raise HTTPException(status_code=422, detail="用户名至少需要3位")
    if len(body.password) < 6:
        raise HTTPException(status_code=422, detail="密码至少需要6位")

    # 查重 — 用户名
    existing = await db.execute(
        select(User).where(User.username == body.username)
    )
    if existing.scalar_one_or_none():
        raise HTTPException(status_code=409, detail="用户名已存在")

    # 查重 — 邮箱
    if body.email:
        existing_email = await db.execute(
            select(User).where(User.email == body.email)
        )
        if existing_email.scalar_one_or_none():
            raise HTTPException(status_code=409, detail="邮箱已被注册")

    user = User(
        username=body.username,
        email=body.email,
        password_hash=hash_password(body.password),
    )
    db.add(user)
    await db.flush()
    return ok(None, "注册成功")


# 内存中存储验证码: {username: {code, expires_at, sent_at, user_id}}
_reset_codes: dict[str, dict] = {}


@router.post("/send-reset-code")
async def send_reset_code(body, db: AsyncSession = Depends(get_db)):
    """
    发送密码重置验证码

    流程:
      1. 校验用户名和邮箱匹配
      2. 频率限制: 同一邮箱 60 秒内不可重复发送 (429)
      3. 生成 6 位随机数字验证码
      4. 通过阿里云 DirectMail 发送 HTML 邮件
      5. 验证码存入内存字典，5 分钟有效期
    """
    import time, random

    result = await db.execute(
        select(User).where(User.username == body.username)
    )
    user = result.scalar_one_or_none()
    if not user or not user.email:
        raise HTTPException(status_code=400, detail="用户名或邮箱不正确")

    # 频率限制
    now = time.time()
    existing = _reset_codes.get(body.username)
    if existing and now - existing.get("sent_at", 0) < 60:
        remaining = int(60 - (now - existing["sent_at"]))
        raise HTTPException(status_code=429, detail=f"请 {remaining} 秒后再试")

    # 生成验证码 + 发送邮件
    code = str(random.randint(100000, 999999))
    _reset_codes[body.username] = {
        "code": code, "expires_at": now + 600, "sent_at": now, "user_id": user.id,
    }
    # ... 发送邮件（调用 email_service.send_email）
    return ok(None, "验证码已发送")

async def create_analysis_task(
    db, user_id, mode, video_url, comment_count, time_range, language,
    analysis_method="llm", background_tasks=None,
) -> AnalysisTask:
    """
    创建分析任务

    步骤:
      1. 解析视频 URL → (platform, platform_video_id)
      2. 查找或创建 Video 记录
      3. 创建 AnalysisTask 记录 (status=queued)
      4. 注册后台爬取任务 (FastAPI BackgroundTasks)
      5. 提前 commit，确保后台任务能在独立 session 中查到该记录
    """
    platform, platform_video_id = parse_video_url(video_url)
    video = await find_video_by_platform_id(db, platform, platform_video_id)

    if video is None:
        # 新视频 → 实时抓取元数据
        if platform == "bilibili":
            info = await fetch_bilibili_video_info(platform_video_id)
        elif platform == "douyin":
            info = await fetch_douyin_video_info(platform_video_id)
        video = await upsert_video(db, info)

    task = AnalysisTask(
        id=str(uuid.uuid4()),
        user_id=user_id, video_id=video.id,
        platform=platform, mode=AnalysisMode(mode),
        comment_limit=comment_count, status=AnalysisStatus.queued,
        analysis_method=analysis_method,
    )
    db.add(task)

    # 注册到 FastAPI BackgroundTasks: 不阻塞当前 HTTP 响应
    if background_tasks is not None:
        background_tasks.add_task(
            run_crawl_task, task_id=task.id, video_id=video.id,
            platform=platform, platform_video_id=platform_video_id,
            comment_limit=comment_count, analysis_method=analysis_method,
        )

    await db.commit()  # 提前提交，保证后台任务能看到此记录
    return task

"""LLM 服务 — 调用 DeepSeek 进行评论分析

核心功能:
  1. analyze_comments_with_llm(): 全量分析（初版分析时调用）

     - 属性情感分析 (4-6个维度): name, positive, negative, neutral
     - 整体综述 (200-400字)

  2. classify_new_comments_with_llm(): 增量分类（实时追踪时调用）

import json, logging

def _build_analysis_prompt(comments: list[dict]) -> str:
    """
    构造全量分析 Prompt

    将评论格式化为 "评论0：xxx\n评论1：xxx..." 形式，
    要求 LLM 通过 comment_indices 数组指回具体评论
    每条评论限制 300 字（防止个别超长评论撑爆 prompt）
    """
    comment_lines = []
    for i, c in enumerate(comments):
        text = c.get("text", "").strip()
        if not text:
            continue
        if len(text) > 300:
            text = text[:300] + "..."
        comment_lines.append(f"评论{i}：{text}")

    comments_text = "\n".join(comment_lines)

    prompt = f"""你是一个专业的评论分析助手。下面是一批用户评论，每条带有编号。

请你完成以下任务，并以 JSON 格式返回结果：

1. 话题分类：将所有评论归纳为若干话题类别（不超过 8 个），每个话题包含：
   - name: 话题名称（简短，如"产品价格讨论"）
   - keywords: 该话题的 3-5 个关键词
   - comment_indices: 属于该话题的评论编号列表
   - summary: 该话题的观点总结（100-200字）
   - sentiment: 情感分布 {{"positive": N, "negative": N, "neutral": N}}

2. 属性情感分析：归纳 4-6 个评价维度（如"画质""玩法""难度"）

3. 整体综述：200-400 字的综合综述

要求：
- 每条评论必须归属到某一个话题（不能遗漏、不能重复归属）
- 情感判断: positive=正面/赞美/支持, negative=负面/批评/反对, neutral=中性/客观/疑问
- 属性维度命名要具体、贴近内容

评论列表：
{comments_text}

请严格按 JSON 格式返回（只返回 JSON，不要附加其他文字）：
{{"topics": [...], "aspects": [...], "overall_summary": "..."}}"""
    return prompt


def _try_repair_json(raw: str) -> dict | None:
    """
    尝试修复被截断的 JSON

    策略1: 直接解析（大多数情况）
    策略2: 截断到最后一个有效 ] 或 } → 补全缺失的 }}]
            至少要有 topics 或 overall_summary 才算修复成功
    """
    try:
        return json.loads(raw)
    except json.JSONDecodeError:
        pass

    try:
        last_valid = max(raw.rfind("]"), raw.rfind("}"))
        if last_valid > 0:
            truncated = raw[:last_valid + 1]
            # 计数缺失的括号数
            open_braces = truncated.count("{") - truncated.count("}")
            open_brackets = truncated.count("[") - truncated.count("]")
            truncated += "}" * max(0, open_braces)
            truncated += "]" * max(0, open_brackets)
            result = json.loads(truncated)
            if result.get("topics") or result.get("overall_summary"):
                logger.info("JSON 修复成功 (截断补全)")
                return result
    except (json.JSONDecodeError, Exception):
        pass

    logger.warning("JSON 修复失败，放弃本次 LLM 结果")
    return None


async def analyze_comments_with_llm(comments: list[dict]) -> dict:
    """
    调用 DeepSeek 大模型全量分析评论

    Args:
        comments: 评论列表，每项 {"text": "评论内容"}

    Returns:
        {"topics": [...], "aspects": [...], "overall_summary": "..."}

    容错: JSON 截断 → 自动修复; API 调用失败 → 返回空结构
    """
    if not comments:
        return {"topics": [], "aspects": [], "overall_summary": ""}

    if not DEEPSEEK_API_KEY:
        logger.error("DEEPSEEK_API_KEY 未配置")
        return {"topics": [], "aspects": [], "overall_summary": ""}

    client = AsyncOpenAI(api_key=DEEPSEEK_API_KEY, base_url=DEEPSEEK_BASE_URL)
    prompt = _build_analysis_prompt(comments)

    try:
        resp = await client.chat.completions.create(
            model=DEEPSEEK_MODEL,
            messages=[
                {"role": "system", "content": "你是一个专业的评论分析助手。请严格按 JSON 格式返回。"},
                {"role": "user", "content": prompt},
            ],
            temperature=0.7,          # 适中的随机性
            max_tokens=16384,         # 足够大的输出空间，避免 JSON 截断
            response_format={"type": "json_object"},  # 强制 JSON 模式

async def classify_new_comments_with_llm(
    new_comments: list[dict], existing_topics: list[dict],
) -> dict:
    """
    增量分类：将追踪期间的新评论归入已有话题 + 逐条情感判断

    Args:
        new_comments: 新评论列表 [{"text": "..."}, ...]
        existing_topics: 已有话题 [{"name": "...", "keywords": [...]}, ...]

    Returns:
        {"assignments": [{"comment_index": 0, "topic_name": "话题名", "sentiment": "positive"}, ...]}

    特点:
      - temperature=0.3 (低温度确保分类一致性)
      - 不匹配已有话题的评论归入"其他"
      - 2 分钟超时（比全量分析快）
    """
    if not new_comments or not DEEPSEEK_API_KEY:
        return {"assignments": []}

    # 构造已有话题描述
    topic_lines = []
    for t in existing_topics:
        kw_str = "、".join(t.get("keywords", []))
        topic_lines.append(f'- "{t["name"]}" [关键词: {kw_str}]')

    # 构造新评论列表
    comment_lines = []
    for i, c in enumerate(new_comments):
        text = c.get("text", "").strip()[:200]  # 限制长度
        comment_lines.append(f"评论{i}：{text}")

    prompt = f"""请将每条新评论分类到最匹配的已有话题中，并判断情感倾向。

已有话题：
{chr(10).join(topic_lines)}

新评论：
{chr(10).join(comment_lines)}

返回格式：
{{"assignments": [{{"comment_index": 0, "topic_name": "话题名", "sentiment": "positive"}}]}}"""

    client = AsyncOpenAI(api_key=DEEPSEEK_API_KEY, base_url=DEEPSEEK_BASE_URL)

    try:
        resp = await client.chat.completions.create(
            model=DEEPSEEK_MODEL,
            messages=[
                {"role": "system", "content": "你是专业的评论分类助手。"},
                {"role": "user", "content": prompt},
            ],
            temperature=0.3,  # 低温度 = 高一致性
            response_format={"type": "json_object"},
            timeout=120,
        )
        result = json.loads(resp.choices[0].message.content)
        return {"assignments": result.get("assignments", [])}
    except Exception as e:
        logger.error("增量分类失败: %s", e)
        return {"assignments": []}


================================================================================
十、聚类分析服务 — Qwen3-Embedding + UMAP + HDBSCAN (backend/service/embed_cluster_service.py)
================================================================================

"""Embedding + 聚类分析 Service

与 LLM 服务并行的第二条分析路径，输出格式完全一致。

技术流程:
  1. 文本向量化: Qwen3-Embedding-0.6B (1024维) → Mean Pooling → L2 归一化
  2. 降维: UMAP (1024维 → 10维)
  3. 聚类: HDBSCAN (基于密度的层次聚类，自动确定簇数)
  4. 关键词提取: jieba 分词 + TF 统计
  5. 情感估算: 内置正负面词表匹配
  6. LLM 精修: 用 DeepSeek 给聚类话题命名 + 写摘要 + 综述
     (失败时回退到规则生成)

核心函数:
  - embed_comments(): 批量文本向量化 (batch_size=32)
  - cluster_embeddings(): UMAP 降维 + HDBSCAN 聚类
  - _extract_keywords(): jieba 分词 + 停用词过滤 + 高频词统计
  - _estimate_sentiment(): 词表匹配估算正/负/中性比例
  - _llm_polish_clusters(): DeepSeek 精修话题名、摘要、综述
  - analyze_comments(): 主入口（同步函数，通过 asyncio.to_thread 包装）
"""

import json, logging, re, numpy as np
from pathlib import Path

logger = logging.getLogger("embed_cluster_service")

# ── Qwen3-Embedding 模型路径 ──
_MODEL_PATH = Path(__file__).resolve().parent.parent.parent / "models" / "embedding" / "models--Qwen--Qwen3-Embedding-0.6B" / "snapshots" / "..."
_model = None  # 全局单例（懒加载）
_tokenizer = None


def _get_model_and_tokenizer():
    """懒加载 Qwen3 模型和分词器（首次调用时才加载，节省内存）"""
    global _model, _tokenizer
    if _model is not None:
        return _model, _tokenizer

    import torch
    from transformers import AutoModel, AutoTokenizer

    tokenizer = AutoTokenizer.from_pretrained(str(_MODEL_PATH), trust_remote_code=True)
    model = AutoModel.from_pretrained(
        str(_MODEL_PATH), trust_remote_code=True, torch_dtype=torch.float32,
    )
    model.eval()  # 评估模式（不计算梯度）
    _model, _tokenizer = model, tokenizer
    return model, tokenizer


def _mean_pooling(last_hidden_state, attention_mask):
    """
    Mean Pooling — 取所有 token 的隐层向量平均值作为句子向量
    相比 CLS token，mean pooling 对长文本更鲁棒
    """
    token_embeddings = last_hidden_state
    mask_expanded = attention_mask.unsqueeze(-1).expand(token_embeddings.size()).float()
    return (token_embeddings * mask_expanded).sum(1) / mask_expanded.sum(1)


def embed_comments(texts: list[str]) -> np.ndarray:
    """
    将评论文本批量向量化

    Args:
        texts: 文本列表

    Returns:
        shape (N, 1024) 的 L2 归一化向量矩阵

    实现细节:
      - batch_size=32（平衡内存和速度）
      - max_length=128（评论通常较短）
      - torch.no_grad() 关闭梯度计算
      - 输出 L2 归一化（余弦相似度 = 内积）
    """
    import torch
    model, tokenizer = _get_model_and_tokenizer()

    all_embeddings = []
    for i in range(0, len(texts), 32):
        batch = texts[i:i + 32]
        inputs = tokenizer(batch, padding=True, truncation=True,
                           max_length=128, return_tensors="pt")
        with torch.no_grad():
            outputs = model(**inputs)
        embedding = _mean_pooling(outputs.last_hidden_state, inputs["attention_mask"])
        embedding = embedding / embedding.norm(dim=1, keepdim=True)  # L2 归一化
        all_embeddings.append(embedding.numpy())

    return np.concatenate(all_embeddings, axis=0)


def cluster_embeddings(vectors: np.ndarray) -> np.ndarray:
    """
    UMAP 降维 + HDBSCAN 聚类

    HDBSCAN 特点:
      - 基于密度，自动确定簇数（不需要预设 K 值）
      - label=-1 表示噪音点（不属于任何簇）
      - min_cluster_size 动态计算: max(3, min(5, N//3))

    参数决策:
      - UMAP n_components=10 (从 1024 维降到 10 维)
      - HDBSCAN min_samples=3, metric=euclidean
    """
    import umap, hdbscan

    n = len(vectors)
    if n < 3:
        return np.zeros(n, dtype=int)  # 太少的数据全部归为同一簇

    min_cluster = max(3, min(5, n // 3))

    # UMAP 降维
    reducer = umap.UMAP(n_components=min(10, n - 2), random_state=42)
    embedding_10d = reducer.fit_transform(vectors)

    # HDBSCAN 聚类
    clusterer = hdbscan.HDBSCAN(
        min_cluster_size=min_cluster, min_samples=3, metric="euclidean",
    )
    labels = clusterer.fit_predict(embedding_10d)

    # 全部为噪音 → 回退为单簇
    if len(set(labels)) - (1 if -1 in labels else 0) == 0:
        labels = np.zeros(n, dtype=int)

    return labels


# ── 情感估算词表 ──
_POSITIVE_WORDS = {"好", "棒", "赞", "喜欢", "爱", "厉害", "优秀", ...}  # 80+ 词
_NEGATIVE_WORDS = {"差", "烂", "垃圾", "恶心", "讨厌", ...}              # 80+ 词
_STOP_WORDS = {"的", "了", "在", "是", "我", ...}                       # 停用词


def _extract_keywords(texts: list[str], top_n: int = 5) -> list[str]:
    """
    从一组文本中提取前 top_n 个高频关键词

    使用 jieba 分词 → 去停用词 → 过滤短词/纯数字/标点 → TF 排序
    """
    import jieba
    word_counts: dict[str, int] = {}

    for t in texts:
        for w in jieba.lcut(t):
            w = w.strip().lower()
            if len(w) < 2 or w in _STOP_WORDS:
                continue
            if re.match(r'^[\d\s\.\,\!\?\.\，\。\！\？\《\》\-]+$', w):
                continue
            word_counts[w] = word_counts.get(w, 0) + 1

    sorted_words = sorted(word_counts.items(), key=lambda x: -x[1])
    return [w for w, _ in sorted_words[:top_n]]


def _estimate_sentiment(texts: list[str]) -> dict:
    """
    基于正负面词表估算情感分布

    规则: 含正面词不含负面词 → positive
          含负面词不含正面词 → negative
          其他 → neutral
    """
    pos = neg = neu = 0
    for t in texts:
        has_pos = any(w in t for w in _POSITIVE_WORDS)
        has_neg = any(w in t for w in _NEGATIVE_WORDS)
        if has_pos and not has_neg:
            pos += 1
        elif has_neg and not has_pos:
            neg += 1
        else:
            neu += 1
    return {"positive": pos, "negative": neg, "neutral": neu}


def analyze_comments(comments: list[dict]) -> dict:
    """
    嵌入 + 聚类分析主入口（同步函数）

    流程:
      1. 提取评论文本 → 2. 向量化 → 3. 聚类分组
      4. 提取关键词 + 估算情感
      5. LLM 精修话题命名 + 综述 → 失败则回退规则生成

    Returns:
      {"topics": [...], "aspects": [...], "overall_summary": "..."}
      与 LLM 路径的 analyze_comments_with_llm() 格式完全一致
    """
    texts = [c.get("text", "").strip() for c in comments if c.get("text", "").strip()]
    if not texts:
        return {"topics": [], "aspects": [], "overall_summary": ""}

    vectors = embed_comments(texts)         # (N, 1024)
    labels = cluster_embeddings(vectors)    # (N,)

    # 按簇分组
    cluster_map: dict[int, list[int]] = {}
    for i, label in enumerate(labels):
        cluster_map.setdefault(label, []).append(i)

    # 排除噪音簇 (label=-1)
    if -1 in cluster_map:
        del cluster_map[-1]

    # 为每个簇提取关键词和情感
    topics = []
    for label, indices in cluster_map.items():
        cluster_texts = [texts[i] for i in indices]
        keywords = _extract_keywords(cluster_texts)
        topic_name = " - ".join(keywords[:2]) if keywords else f"话题 {label+1}"
        sentiment = _estimate_sentiment(cluster_texts)
        topics.append({
            "name": topic_name, "keywords": keywords,
            "comment_indices": indices, "sentiment": sentiment,
            "summary": f"共 {len(indices)} 条，关键词：{'、'.join(keywords)}",
        })

    topics.sort(key=lambda t: -len(t["comment_indices"]))  # 按评论数降序

    # LLM 精修 → 规则兜底
    try:
        topics, overall_summary, aspects = _llm_polish_clusters(topics, texts, len(texts))
    except Exception:
        overall_summary = _generate_overall_summary(topics, len(texts))
        aspects = _build_aspects_from_clusters(topics)

    return {"topics": topics, "aspects": aspects, "overall_summary": overall_summary}

十二、实时追踪服务 (backend/service/tracking_service.py)
================================================================================

"""追踪任务 Service — 增量评论采集 + 增量 LLM 分类 + 积分消耗

核心设计:
  1. 内存注册表 _active_trackings: tracking_id → asyncio.Task
     统一管理所有运行中的追踪协程

  2. 轮询循环 poll_tracking_comments():
     - 每 10 秒检查一次新评论
     - 增量抓取（B站: 按时间排序逐页检查，遇到已知 last_comment_id 停止）
     - 清洗 + 写入 DB + 增量 LLM 分类

  3. 积分系统:
     - 按实际经过时间 × 积分率消耗积分
     - 积分不足 → 自动标记 exhausted 状态

  4. 生命周期:
     - resume_all_active_trackings(): 服务启动时恢复 DB 中 status=active 的任务
     - shutdown_all_trackings(): 服务关闭时取消所有轮询 + 标记 DB 为 stopped

  5. 容错:
     - 连续失败 5 次 → 自动停止
     - LLM 分类连续失败 10 次 → 停止分类但继续轮询
     - 4 层退出保护: cancel / DB stopped / 注册表移除
"""

import asyncio, json, logging, uuid
from datetime import datetime, timedelta

# 模块级状态
_active_trackings: dict[str, asyncio.Task] = {}  # 活跃追踪注册表
DEFAULT_POLL_INTERVAL = 10       # 默认轮询间隔（秒）
MAX_POLL_PAGES = 5               # 单轮最多抓取页数
MAX_CONSECUTIVE_FAILURES = 5     # 连续失败次数上限
ANALYSIS_INTERVAL_SECONDS = 60   # 最小 LLM 分析间隔
MAX_ANALYSIS_FAILURES = 10       # LLM 分析连续失败上限


async def poll_tracking_comments(tracking_id: str) -> None:
    """
    主轮询协程

    每轮流程:
      1. 检查注册表 → 不在则退出
      2. 加载追踪任务 + 关联数据
      3. 积分消耗计算（按实际经过时间）
      4. 检查积分余额 → 不足则停止
      5. 增量抓取新评论（直到遇到 last_comment_id）
      6. 清洗 → 写入 DB → 增量 LLM 分类
      7. 条件触发周期性报告邮件
    """
    consecutive_failures = 0
    consecutive_analysis_failures = 0

    while True:
        # 三重保护
        if tracking_id not in _active_trackings:
            break

        try:
            async with get_sessionmaker()() as db:
                tracking = await db.get(TrackingTask, tracking_id)
                if tracking is None or tracking.status != TrackingStatus.active:
                    break

                # 积分消耗
                now = datetime.utcnow()
                elapsed_hours = (now - tracking.last_poll_at).total_seconds() / 3600
                credit_cost = elapsed_hours * tracking.credits_rate_per_hour

                if user.credits < credit_cost:
                    tracking.status = TrackingStatus.exhausted
                    break
                user.credits -= int(credit_cost)

                # 增量抓取新评论
                new_raw = await _fetch_new_comments(
                    platform, fetch_id, tracking.last_comment_id, extras
                )

                if new_raw:
                    # 写入 → 清洗 → 标记 → 更新游标
                    await batch_upsert_comments(db, video_id, platform, new_raw, ...)
                    cleaned, _ = clean_comments(new_raw)
                    await mark_comments_cleaned(db, video_id, cleaned_cids, ...)
                    tracking.last_comment_id = new_raw[-1].get("cid", "")

                # 条件触发增量 LLM 分类（至少间隔 60 秒）
                if time_since_last >= ANALYSIS_INTERVAL_SECONDS:
                    unanalyzed = await _get_unanalyzed_cleaned_comments(...)
                    if unanalyzed:
                        llm_result = await classify_new_comments_with_llm(
                            comments_for_llm, existing_topics
                        )
                        await _apply_classification_results(
                            db, unanalyzed, llm_result, existing_topics, ...
                        )

                await db.commit()

        except asyncio.CancelledError:
            break
        except Exception as e:
            consecutive_failures += 1
            if consecutive_failures >= MAX_CONSECUTIVE_FAILURES:
                tracking.status = TrackingStatus.stopped
                break

        await asyncio.sleep(DEFAULT_POLL_INTERVAL)

    finally:
        _active_trackings.pop(tracking_id, None)


def start_tracking(tracking_id: str) -> None:
    """
    启动追踪轮询 — 创建 asyncio.Task 并注册到内存字典
    在 FastAPI 主事件循环中调用
    """
    if tracking_id in _active_trackings:
        return  # 已在运行中

    task = asyncio.create_task(poll_tracking_comments(tracking_id))
    _active_trackings[tracking_id] = task


async def resume_all_active_trackings() -> int:
    """
    服务启动时恢复所有 active 状态的追踪任务

    安全机制:
      - last_poll_at 超过 5 分钟 → 视为异常遗留，标记 stopped
      - 从未轮询过 + started_at 超过 5 分钟 → 也标记 stopped
      - 仅恢复最近的活跃任务
    """
    cutoff = datetime.utcnow() - timedelta(minutes=5)

    async with get_sessionmaker()() as db:
        result = await db.execute(
            select(TrackingTask).where(TrackingTask.status == TrackingStatus.active)
        )
        all_tasks = result.scalars().all()

        resumed = 0
        for task in all_tasks:
            if task.id in _active_trackings:
                continue
            # 过期遗留 → stopped
            if task.last_poll_at and task.last_poll_at < cutoff:
                task.status = TrackingStatus.stopped
                continue
            start_tracking(task.id)
            resumed += 1

        await db.commit()
    return resumed


async def shutdown_all_trackings():
    """
    服务关闭时取消所有轮询 + 更新 DB 状态

    重要: 必须更新 DB，否则下次启动时 resume 会错误恢复已关闭的任务
    """
    for tracking_id, task in list(_active_trackings.items()):
        task.cancel()
    _active_trackings.clear()

    # 将 DB 中所有 active 标记为 stopped
    await db.execute(
        update(TrackingTask)
        .where(TrackingTask.status == TrackingStatus.active)
        .values(status=TrackingStatus.stopped, stopped_at=datetime.utcnow())
    )
    await db.commit()

/**
 * API Module — 所有后端接口的统一封装
 *
 * 设计要点:
 *   1. 全部接口调用经过 request() 统一处理
 *   2. 自动携带 JWT token (Authorization: Bearer xxx)
 *   3. 统一错误处理: HTTP 错误 / 业务错误 (code !== 0) / 网络错误
 *   4. 响应格式: {code: 0, message: "ok", data: ...}
 *      - code === 0  → 返回 data 字段
 *      - code !== 0  → 抛出 Error(message)
 */

const API_BASE = '/api';

/** 从 localStorage 读取 JWT token */
function getToken() {
  return localStorage.getItem('token') || '';
}

/**
 * 统一 HTTP 请求封装
 *
 * @param {string} method - HTTP 方法 (GET/POST/PUT/DELETE)
 * @param {string} path   - 接口路径，如 '/auth/login'
 * @param {object} [body] - 请求体（JSON），GET 请求省略
 * @returns {Promise<any>} 响应中的 data 字段
 * @throws {Error} 网络错误 / HTTP 错误 / 业务错误
 */
async function request(method, path, body) {
  const opts = {
    method,
    headers: { 'Authorization': 'Bearer ' + getToken() }
  };
  if (body) {
    opts.headers['Content-Type'] = 'application/json';
    opts.body = JSON.stringify(body);
  }

  let res;
  try {
    res = await fetch(API_BASE + path, opts);
  } catch (e) {
    throw new Error('网络错误，无法连接服务器');
  }

  // HTTP 错误 → 尝试读取后端错误信息
  if (!res.ok) {
    let detail = '请求失败 (HTTP ' + res.status + ')';
    try {
      const errJson = await res.json();
      if (errJson.message) detail = errJson.message;
      else if (typeof errJson.detail === 'string') detail = errJson.detail;
    } catch (_) {}
    throw new Error(detail);
  }

  // 解析 JSON，检查业务错误码
  const json = await res.json();
  if (json.code !== 0) {
    throw new Error(json.message || '请求失败');
  }
  return json.data;
}

/** GET 简写 */
function get(path) { return request('GET', path); }
/** POST 简写 */
function post(path, body) { return request('POST', path, body); }
/** PUT 简写 */
function put(path, body) { return request('PUT', path, body); }

/**
 * 格式化大数字 — 万 / 亿
 * 120000 → "12.0万", 150000000 → "1.5亿"
 */
function formatCount(n) {
  if (!n && n !== 0) return '—';
  n = parseInt(n, 10);
  if (isNaN(n)) return '—';
  if (n >= 100000000) return (n / 100000000).toFixed(1) + '亿';
  if (n >= 10000) return (n / 10000).toFixed(1) + '万';
  return n.toLocaleString();
}

/**
 * 格式化秒数为可读时长
 * 3661 → "1h 01m", 90 → "01:30"
 */
function formatDuration(seconds) {
  const h = Math.floor(seconds / 3600);
  const m = Math.floor((seconds % 3600) / 60);
  const s = seconds % 60;
  if (h > 0) return h + 'h ' + m.toString().padStart(2, '0') + 'm';
  return m.toString().padStart(2, '0') + ':' + s.toString().padStart(2, '0');
}

    }

    // 加载顶部统计
    if (options && options.loadHeaderStats) {
      result.headerStats = await api.getHeaderStats();
      setText('stat-today-comments', result.headerStats.today_comments.toLocaleString());
      setText('stat-new-videos', result.headerStats.new_videos.toString());
      setText('stat-completion', result.headerStats.analysis_completion_rate + '%');
      setText('stat-hot-topic', result.headerStats.hot_topic || '—');
    }

    // 检查追踪状态（控制侧边栏红点）
    const trackingStatus = await api.hasActiveTracking();
    const navDot = document.getElementById('sidebar-tracking-dot');
    if (navDot && trackingStatus.has_active) {
      navDot.style.display = '';
    }
  } catch (e) {
    console.error('initPage error:', e);
  }

  return result;
}

"""邮件发送 Service — 阿里云 Direct Mail

对接阿里云邮件推送 API，支持:
  - 密码重置验证码邮件
  - 分析报告 HTML 邮件（手动发送 + 自动发送 + 周期推送）
"""

import json, logging
from aliyunsdkcore.client import AcsClient
from aliyunsdkcore.request import CommonRequest
from backend.config import (
    ALIYUN_ACCESS_KEY_ID, ALIYUN_ACCESS_KEY_SECRET,
    ALIYUN_ACCOUNT_NAME, ALIYUN_REGION,
)

logger = logging.getLogger("email_service")


def send_email(to_address: str, subject: str, html_body: str) -> bool:
    """
    通过阿里云 Direct Mail 发送 HTML 邮件

    Args:
        to_address: 收件人邮箱地址
        subject:    邮件主题
        html_body:  HTML 格式邮件正文

    Returns:
        True → 发送成功, False → 发送失败

    实现流程:
      1. 创建 AcsClient 进行身份鉴权
      2. 构造 CommonRequest → SingleSendMail 接口
      3. 填充 AccountName (发信账号), ToAddress (收件人), Subject, HtmlBody
      4. 调用 do_action_with_exception → 记录 RequestId 日志
    """
    if not ALIYUN_ACCESS_KEY_ID:
        logger.error("ALIYUN_ACCESS_KEY 未配置，无法发送邮件")
        return False

    client = AcsClient(
        ALIYUN_ACCESS_KEY_ID, ALIYUN_ACCESS_KEY_SECRET, ALIYUN_REGION
    )

    request = CommonRequest()
    request.set_domain("dm.aliyuncs.com")
    request.set_version("2015-11-23")
    request.set_action_name("SingleSendMail")
    request.set_method("POST")

    request.add_query_param("AccountName", ALIYUN_ACCOUNT_NAME)
    request.add_query_param("AddressType", "1")
    request.add_query_param("ReplyToAddress", "false")
    request.add_query_param("ToAddress", to_address)
    request.add_query_param("Subject", subject)
    request.add_query_param("HtmlBody", html_body)

    try:
        response = client.do_action_with_exception(request)
        result = json.loads(response)
        logger.info("邮件发送成功 -> %s, RequestId=%s",
                     to_address, result.get("RequestId"))
        return True
    except Exception as e:
        logger.error("邮件发送失败 -> %s: %s", to_address, e)
        return False

【系统架构总结】
================================================================================

一、后端分层架构:
   API 路由层 (backend/api/)        ← 请求接收、参数校验、路由分发
        ↓
   业务逻辑层 (backend/service/)    ← 核心业务处理、算法调度
        ↓
   数据访问层 (backend/models/)     ← ORM 模型定义 + SQLAlchemy 2.0 异步会话
        ↓
   MySQL 8.0 数据库

二、分析流水线 (Pipeline):
   创建任务 → 抓取评论 → 清洗数据 → 双路径分析:
     ├─ LLM 路径: DeepSeek 话题分类 + 情感分析 + 综述生成
     └─ Cluster 路径: Qwen3-Embedding 向量化 → UMAP 降维 → HDBSCAN 聚类
   → 写入 Topic 表 → 自动发送邮件报告

三、实时追踪系统:
   分析完成 → 创建 TrackingTask → asyncio 后台轮询
   → 增量抓取新评论 → 增量 LLM 分类 → 更新话题数据
   → 周期性发送更新报告邮件 → 积分实时扣减
   → 支持服务重启后自动恢复

四、前端架构:
   Vue.js 3 (Composition API + script setup) + Chart.js 可视化
   → SPA 单页面应用 (7 个核心页面)
   → api.js 统一 HTTP 请求 + JWT 鉴权
   → charts.js 统一图表管理 (6 种图表类型)
   → common.js 全局状态 + UI 工具函数
   → SSE 实时进度推送 (替代传统轮询)
   → 深色主题 (黑底 + 橙/青强调色)

五、安全设计:
   - bcrypt 加盐密码哈希
   - JWT (HS256, 24h 过期, "记住我" 30 天)
   - 依赖注入鉴权中间件 (get_current_user)
   - 前端路由守卫 + 401 响应拦截
   - 路径遍历攻击防护
   - 验证码发送频率限制 (60s) + 过期机制 (5min)

六、容错与降级:
   - LLM JSON 截断自动修复算法
   - 聚类分析 LLM 精修 → 规则兜底
   - SSE 鉴权错误通过事件传递（而非 HTTP 状态码）
   - 追踪任务连续失败自动停止
   - 前端全状态覆盖: 骨架屏 → 空数据 → 错误提示 → 重试
   - 图表重绘前先销毁旧实例（防 Canvas 内存泄漏）
